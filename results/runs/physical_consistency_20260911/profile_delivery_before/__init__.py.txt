@dataclass(frozen=True)
class Model:
    nodes: int = 50
    waves: int = 10
    bits: int = 2048
    def __post_init__(self):
        if self.nodes < 1 or self.waves < 1 or self.bits < 64:
            raise ValueError('Positive node/wave counts and explicit precision required')
    @property
    def coefficients(self):
        m = self.nodes
        return 1 + 2*m + m*m + m*(m+1)//2
    @property
    def physical_rows(self):
        return 3*self.nodes*self.waves
    @property
    def shape(self):
        return 2*self.physical_rows + 11, self.coefficients

@dataclass(frozen=True)
class UVConfig:
    moment_source: str = 'printed'
    sr_error: str = 'raw-absolute'
    cutoff: str = 'hard-midpoint'
    mq_rule: str = 'arithmetic-mean'
    pion_mass_gev: float = .140
    matching_energy_gev: float = 1.2
    up_mass_gev: float = .004
    down_mass_gev: float = .0073
    alpha_s: float = .4
    gluon_condensate_gev4: float = .023
    scalar_condensate_gev4: float = -.0001
    sr_tolerance: float = .002
    ff_tolerance: float = .00006
    def __post_init__(self):
        choices = {'moment_source': ('printed', 'eq250'),
                   'sr_error': ('raw-absolute', 'normalized-absolute'),
                   'cutoff': ('hard-midpoint', 'clipped-phi'),
                   'mq_rule': ('arithmetic-mean', 'rms')}
        for name, allowed in choices.items():
            if getattr(self, name) not in allowed:
                raise ValueError(f'{name} must be one of {allowed}')
        for name in ('pion_mass_gev', 'matching_energy_gev', 'up_mass_gev',
                     'down_mass_gev', 'alpha_s', 'sr_tolerance', 'ff_tolerance'):
            if not isfinite(getattr(self, name)) or getattr(self, name) <= 0:
                raise ValueError(f'{name} must be finite and positive')
        if not all(isfinite(v) for v in (self.gluon_condensate_gev4, self.scalar_condensate_gev4)):
            raise ValueError('Condensates must be finite, with their physical signs retained')
        published_inputs = (self.pion_mass_gev, self.matching_energy_gev, self.up_mass_gev,
            self.down_mass_gev, self.alpha_s, self.gluon_condensate_gev4, self.scalar_condensate_gev4)
        if self.moment_source == 'printed' and published_inputs != (.140, 1.2, .004, .0073, .4, .023, -.0001):
            raise ValueError('Changed physical inputs require eq250; printed moments belong to v3 inputs')
        if self.s0 <= 4:
            raise ValueError('The FESR cutoff must exceed the two-pion threshold')
    @property
    def s0(self):
        return (self.matching_energy_gev/self.pion_mass_gev)**2
    @property
    def mq_squared(self):
        u, d = self.up_mass_gev, self.down_mass_gev
        return ((u+d)**2/4 if self.mq_rule == 'arithmetic-mean' else
                (u*u+d*d)/2)/self.pion_mass_gev**2
def solve_feasible_basis_lp(costs,matrix,target,basic_columns,initial,seconds):
    import highspy
    from types import SimpleNamespace
    matrix=matrix.tocsc();m,n=matrix.shape;h=highspy.Highs();lp=highspy.HighsLp()
    if len(set(map(int,basic_columns)))!=m:raise ValueError('One independent basic column per equality required')
    lp.num_col_,lp.num_row_=n,m;lp.col_cost_=costs
    lp.col_lower_,lp.col_upper_=np.zeros(n),np.full(n,highspy.kHighsInf)
    lp.row_lower_,lp.row_upper_=target,target
    lp.a_matrix_.format_=highspy.MatrixFormat.kColwise;lp.a_matrix_.num_col_,lp.a_matrix_.num_row_=n,m
    lp.a_matrix_.start_,lp.a_matrix_.index_,lp.a_matrix_.value_=matrix.indptr,matrix.indices,matrix.data
    for name,value in [('output_flag',False),('presolve','off'),('solver','simplex'),('simplex_strategy',4),('threads',8),('time_limit',float(seconds)),('primal_feasibility_tolerance',1e-9),('dual_feasibility_tolerance',1e-9)]:h.setOptionValue(name,value)
    if h.passModel(lp)==highspy.HighsStatus.kError:raise ArithmeticError('HiGHS rejected the LP data')
    basis=highspy.HighsBasis();statuses=[highspy.HighsBasisStatus.kLower]*n
    for column in basic_columns:statuses[int(column)]=highspy.HighsBasisStatus.kBasic
    basis.col_status=statuses;basis.row_status=[highspy.HighsBasisStatus.kLower]*m
    if h.setSolution(n,np.arange(n,dtype=np.int32),initial)==highspy.HighsStatus.kError or h.setBasis(basis)==highspy.HighsStatus.kError:raise ArithmeticError('HiGHS rejected the feasible starting basis')
    h.run();status=h.getModelStatus();solution=h.getSolution();info=h.getInfo()
    valid=solution.value_valid and np.isfinite(solution.col_value).all();x=np.asarray(solution.col_value if valid else initial,float)
    return SimpleNamespace(x=x,fun=float(np.dot(costs,x)),success=status==highspy.HighsModelStatus.kOptimal,
        status=int(status),message=h.modelStatusToString(status),nit=info.simplex_iteration_count,
        initial_objective=float(np.dot(costs,initial)),primal_infeasibilities=info.num_primal_infeasibilities,
        used_initial_fallback=not valid,solver_version=h.version(),row_dual=np.asarray(solution.row_dual,float))
def extended_triangular(matrix, rhs, transpose=False):
    x=np.asarray(rhs,dtype=np.longdouble).copy()
    for i in (range(len(x)) if transpose else range(len(x)-1,-1,-1)):
        x[i]=(x[i]-(matrix[:i,i]@x[:i] if transpose else matrix[i,i+1:]@x[i+1:]))/matrix[i,i]
    return x
def extended_sparse_blocks(matrix):
    from scipy.sparse import csr_matrix
    csr=csr_matrix(matrix,dtype=np.longdouble)
    edges=np.unique(np.r_[0,np.searchsorted(csr.indptr,np.arange(1,8)*csr.nnz/8),csr.shape[0]])
    blocks=[]
    for lo,hi in zip(edges[:-1],edges[1:]):
        a,b=csr.indptr[[lo,hi]]
        block=csr_matrix((csr.data[a:b],csr.indices[a:b],csr.indptr[lo:hi+1]-a),shape=(hi-lo,csr.shape[1]),copy=False)
        blocks.append((lo,hi,block,block.T))
    return tuple(blocks)
def extended_product(matrix,vector,transpose=False,pool=None):
    if isinstance(matrix,tuple):
        action=lambda block:block[3]@vector[block[0]:block[1]] if transpose else block[2]@vector
        parts=map(action,matrix) if pool is None else pool.map(action,matrix)
        return np.sum(list(parts),axis=0,dtype=np.longdouble) if transpose else np.concatenate(list(parts))
    action=lambda pair:np.einsum('ij,i->j',*pair) if transpose else pair[0]@pair[1]
    if pool is None:return action((matrix,vector))
    step=max(1,(len(matrix)+7)//8)
    parts=pool.map(action,((matrix[j:j+step],vector[j:j+step] if transpose else vector) for j in range(0,len(matrix),step)))
    return np.sum(list(parts),axis=0,dtype=np.longdouble) if transpose else np.concatenate(list(parts))
def support_gradient_width(A,kap,error,M,L,free,bound):
    from .quotient import raw_support_residual
    n=(len(A)-11)//2;zero=np.zeros(n,np.longdouble)
    kr,ki,r=raw_support_residual(A,kap,zero,zero,np.zeros(8),error,M,L,free)

def radial_coordinates(F,c,rows,free,density_end=None):
    """Isolate stiff constraint features in free/current increment coordinates; rho is unchanged."""
    end=F.shape[1] if density_end is None else density_end
    eligible=np.r_[np.arange(free),np.arange(end,F.shape[1])]
    candidates=list(rows)
    if density_end is not None:
        score=np.max(abs(F[:,free:end]),axis=1);candidates+=np.argsort(score)[-48:][::-1].tolist()
    chosen=[];pivots=[];echelon=[];vectors=[];units=[]
    for i in dict.fromkeys(candidates):
        unit=np.max(abs(F[i]))
        if unit==0:continue
        u=F[i]/unit;v=u.copy()
        for row,j in zip(echelon,pivots):v-=v[j]*row
        j=int(eligible[np.argmax(abs(v[eligible]))])
        if abs(v[j])<1e-6:continue
        chosen.append(i);pivots.append(j);echelon.append(v/v[j]);vectors.append(u);units.append(unit)
        if len(chosen)==12:break
    if not chosen:return F,c,lambda w:w
    U=np.asarray(vectors);j=np.asarray(pivots);B=U[:,j];k=len(j);T=np.column_stack((B,np.eye(k,dtype=np.longdouble)))
    for i in range(k):
        at=i+int(np.argmax(abs(T[i:,i])));T[[i,at]]=T[[at,i]];T[i]/=T[i,i]
        for r in range(k):
            if r!=i:T[r]-=T[r,i]*T[i]
    inverse=T[:,k:];P=F[:,j]@inverse;matrix=F-P@U;matrix[:,j]=P
    matrix[chosen]=0;matrix[np.asarray(chosen),j]=units
    v=c[j]@inverse;cost=c-v@U;cost[j]=v
    def restore(w):
        remainder=w.copy();remainder[j]=0;z=w.copy();z[j]=inverse@(w[j]-U@remainder);return z
    return matrix,cost,restore

def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()
def fraction_ball(value):
    value = Fraction(value)
    return arb(value.numerator)/value.denominator
def encode_real_ball(value):
    if value.is_zero():
        return None
    if not value.is_finite():
        raise ValueError('Finite enclosures required')
    m, e = value.mid().man_exp()
    r, re = value.rad().man_exp()
    return [[hex(int(m)), int(e)], [hex(int(r)), int(re)]]

def decode_real_ball(record):
    if record is None:
        return arb(0)
    return arb((int(record[0][0], 16), record[0][1]),
               (int(record[1][0], 16), record[1][1]))

def read_json(path, expected_hash=None):
    path = Path(path)
    if expected_hash is not None and digest(path) != expected_hash:
        raise ValueError('Input hash mismatch: '+str(path))
    with (gzip.open(path, 'rt') if path.suffix == '.gz' else path.open()) as stream:
        return json.load(stream)

def write_json(path, value):
    path = Path(path)
    temporary = Path(str(path)+'.partial')
    temporary.write_text(json.dumps(value, indent=2,default=json_scalar)+'\n')
    temporary.replace(path)

def snapshot_core(directory):
    directory = Path(directory)
    directory.mkdir(parents=True, exist_ok=False)
    package = Path(__file__).parent
    files = sorted(package.glob('*.py'))
    sources = {p.name: p.read_text() for p in files}
    hashes = {p.name: digest(p) for p in files}
    write_json(directory/'source.json', dict(source_snapshot_utf8=sources, source_sha256=hashes))
    return hashes

def json_scalar(value):
    if isinstance(value,np.generic):return float(value) if isinstance(value,np.floating) else value.item()
    if isinstance(value,np.ndarray):return value.tolist()
    raise TypeError('Unsupported JSON value: '+type(value).__name__)

def _phase_data(path):
    from .analysis import phase_shifts
    d=read_json(Path(path)/'evaluation.json');s=np.asarray(d['energies'],float)
    expected=d.get('coefficient_sha256')
    if expected and (digest(Path(d['coefficients']))!=expected or d.get('selection',{}).get('coefficient_sha256',expected)!=expected):raise ValueError('Evaluated/selected amplitude identity changed')
    for name,suffix in (('current','json'),('joint','npz')):
        expected=d.get('selection',{}).get(name+'_sha256')
        if expected and digest(Path(d['coefficients']).parent/(name+'.'+suffix))!=expected:raise ValueError('Selected joint point changed after evaluation')
    if d['waves']!=[[0,0],[2,0],[1,1]] or d['unitarity']['status']!='sampled_passed':raise ValueError('Passed native S0/S2/P1 evaluation required')
    f=np.asarray(d['f']);delta,eta=phase_shifts(f[:,:,0]+1j*f[:,:,1],s)
    return d,.140*np.sqrt(s),np.degrees(delta),eta
