def chiral_slices(chiral_norm='separate-l2'):
    if chiral_norm=='separate-l2':return (slice(0,4),slice(4,8))
    if chiral_norm=='combined-l2':return (slice(0,8),)
    raise ValueError('Chiral norm must be separate-l2 or combined-l2')

def declared_chiral_norm(args):
    """Optimization must name its uncertainty set; never infer it from plots."""
    required=(args.command in ('boundary','dual') and args.mode!='pure' or
              args.command=='regions' or args.command=='resolution' and not args.profile_runs)
    if args.chiral_norm is None and required:
        raise ValueError('Declare --chiral-norm explicitly; the paper does not identify a unique norm')
    return args.chiral_norm or 'separate-l2'

def weinberg_waves(s,fpi_over_mpi=92/140):
    s=np.asarray(s,dtype=float)
    return np.stack((2*s-1,2-s,(s-4)/3),axis=-1)/(16*np.pi**2*fpi_over_mpi**2)

def phase_space(s):
    s = arb(s)
    if not s.is_finite() or not s >= 4:
        raise ValueError('Physical scattering energy must be finite and at least 4')
    return (1-4/s).sqrt()

def scattering_matrix(f, s):
    f = acb(f)
    if not f.is_finite():
        raise ValueError('A finite partial wave is required')
    return 1 + acb(0, 1)*arb.pi()*phase_space(s)*f

def unitarity_margin(f, s):
    f = acb(f)
    if not f.is_finite():
        raise ValueError('A finite partial wave is required')
    kappa = arb.pi()*phase_space(s)
    return kappa*(2*f.imag-kappa*(f.real*f.real+f.imag*f.imag))

def scattering_gram(f, s):
    S = scattering_matrix(f, s)
    return acb_mat([[1, S], [S.conjugate(), 1]])

def current_kinematic_squares(s):
    beta = phase_space(s)
    return 3*beta/(256*arb.pi()**5), (s-4)*beta/(384*arb.pi()**5)

def current_gram(S, F, rho, k_squared):
    if not k_squared > 0:
        raise ValueError('The current kinematic factor must be positive')
    S, F = acb(S), acb(F)
    return acb_mat([[1, S, F], [S.conjugate(), 1, F.conjugate()],
                    [F.conjugate(), F, rho/k_squared]])

def chiral_bound(mode, tolerance):
    if mode not in ('pure', 'chiral'):
        raise ValueError('Support mode must be pure or chiral')
    if mode == 'pure':
        return None
    if not isfinite(tolerance) or tolerance <= 0:
        raise ValueError('A finite positive chiral tolerance is required')
    return float(tolerance)

def fesr_weights(M, config=UVConfig(), *, normalized=True):
    from .quotient import _fesr_grid
    s, widths, jacobian = _fesr_grid(M, config)
    powers = np.array(FESR_POWERS)[:, None]
    weights = widths*jacobian*s**powers
    return weights/config.s0**(powers+2) if normalized else weights

def fesr_data(M, config=UVConfig()):
    from .quotient import _fesr_grid, fesr_targets
    from dataclasses import asdict
    s, widths, _ = _fesr_grid(M, config)
    scales = config.s0**(np.array(FESR_POWERS)+2)
    errors = np.full(4, config.sr_tolerance)
    raw_errors = errors if config.sr_error == 'raw-absolute' else errors*scales
    return dict(config=asdict(config), s0=config.s0, nodes=s.tolist(),
        channels=list(FESR_CHANNELS), powers=list(FESR_POWERS),
        spectral_density_gev_powers={'S0': 4, 'P1': 2}, raw_moment_gev_powers=[6, 8, 2, 4],
        form_factor_gev_powers={'S0': 2, 'P1': 0}, units='divide physical GeV quantities by m_pi to the stated power',
        targets_normalized=fesr_targets(config).tolist(),
        targets_raw=fesr_targets(config, normalized=False).tolist(),
        weights_raw=fesr_weights(M, config, normalized=False).tolist(),
        weights_normalized=fesr_weights(M, config).tolist(),
        tolerances_raw=raw_errors.tolist(), tolerances_normalized=(raw_errors/scales).tolist(),
        error_grouping='four independent absolute scalar residuals',
        cell_phi_widths=widths.tolist(), high_energy_mask=(s > config.s0).tolist(),
        form_factors_at_zero=[1., 1.], mq_squared=config.mq_squared,
        rescaled_ff_caps_squared=[2*config.mq_squared*config.ff_tolerance, config.ff_tolerance/2],
        ff_cap_variable='abs(mathcal F)^2 = k_squared * abs(F)^2',
        finite_prescription_is_conditional=True)

def current_operators(H,kappa,energies,waves,M,config=None,bits=256):
    from scipy.sparse import lil_matrix
    from flint import ctx
    from .operators import midpoint_grid
    from .kernels import pv_matrix, density_labels
    config=UVConfig() if config is None else config
    p=len(density_labels(M));N=len(kappa);total=p+4*M
    H=np.asarray(H);kap=np.asarray(kappa,float);ss=np.asarray(energies,float);ww=np.asarray(waves)
    if H.shape!=(2*N+11,p) or kap.shape!=(N,) or ss.shape!=(N,) or ww.shape!=(N,2):
        raise ValueError('Complete original amplitude rows and matching physical metadata required')
    if type(bits) is not int or bits<64:raise ValueError('Explicit current arithmetic precision >=64 required')
    with ctx.workprec(bits):
        nodes=np.array([float(x) for x in midpoint_grid(M)[0]])
        kernel=pv_matrix(M);K=np.array([[float(kernel[i,j]) for j in range(M)] for i in range(M)])
        indices=np.empty((2,M),dtype=int)
        for channel in range(2):
            for j,s in enumerate(nodes):
                matches=np.flatnonzero(np.isclose(ss,s,rtol=64*np.finfo(float).eps,atol=0)&
                    (ww[:,0]==channel)&(ww[:,1]==channel))
                if not len(matches):raise ValueError('Every native S0/P1 node must be present in the same amplitude')
                indices[channel,j]=matches[0]
        actual=ss[indices[0]]
        factors=[current_kinematic_squares(float(s)) for s in actual]
        k2=np.array([[float(pair[ch]) for pair in factors] for ch in range(2)])
        k=np.array([[float(pair[ch].sqrt()) for pair in factors] for ch in range(2)])
    selected=np.r_[indices.ravel(),N+indices.ravel()]
    if not np.isfinite(H[selected]).all() or not np.isfinite(k).all() or not np.all(k>0):
        raise ValueError('Finite current rows strictly above threshold required')
    expected=np.pi*np.sqrt(1-4/ss[indices])
    if not np.allclose(kap[indices],expected,rtol=64*np.finfo(float).eps,atol=0):
        raise ValueError('Current blocks require the original kappa=pi*sqrt(1-4/s)')
    gram=lil_matrix((12*M,total));constant=np.zeros(12*M);uv=fesr_data(M,config)
    for ch in range(2):
        start=p+ch*M;rho=p+(2+ch)*M
        for j,index in enumerate(indices[ch]):
            r=6*(ch*M+j);ki=k[ch,j];kapj=kap[index]
            gram[r,:p]=-kapj*H[N+index];gram[r+1,:p]=np.sqrt(2.)*kapj*H[index]
            gram[r+2,:p]=kapj*H[N+index];gram[r+3,start:start+M]=2*ki*K[j]
            gram[r+4,start+j]=2*ki;gram[r+5,rho+j]=1
            constant[r]=2;constant[r+3]=2*ki
    moments=lil_matrix((4,total));weights=np.asarray(uv['weights_raw'])
    for a in range(4):moments[a,p+(2+a//2)*M:p+(3+a//2)*M]=weights[a]
    high=np.flatnonzero(uv['high_energy_mask']);count=2*len(high)
    caps=lil_matrix((3*count,total));cap_constant=np.zeros(3*count)
    for ch in range(2):
        for a,j in enumerate(high):
            r=3*(ch*len(high)+a);start=p+ch*M;ki=k[ch,j]
            caps[r+1,start:start+M]=ki*K[j];caps[r+2,start+j]=ki
            cap_constant[r]=np.sqrt(uv['rescaled_ff_caps_squared'][ch]);cap_constant[r+1]=ki
    slices={'C_flat':[0,p],**{name:[p+a*M,p+(a+1)*M] for a,name in
        enumerate(('ImF0','ImF1','rho_current0','rho_current1'))}}
    return dict(matrices=dict(gram_linear=gram.tocsr(),moment_linear=moments.tocsr(),ff_cap_linear=caps.tocsr()),
        arrays=dict(gram_constant=constant,ff_cap_constant=cap_constant,hilbert_kernel=K,nodes=actual,
            k_squared=k2,k=k,native_scattering_indices=indices,high_energy_indices=high,
            moment_targets_raw=np.asarray(uv['targets_raw']),moment_errors_raw=np.asarray(uv['tolerances_raw']),
            form_factors_at_zero=np.asarray(uv['form_factors_at_zero']),ff_caps_squared=np.asarray(uv['rescaled_ff_caps_squared'])),
        metadata=dict(M=M,amplitude_coefficients=p,total_variables=total,variable_slices=slices,bits=bits,
            affine_convention='slack=constant+linear@variables; conic A=-linear, b=constant',
            gram_svec_order=['G11','sqrt(2)*G12','G22','sqrt(2)*G13','sqrt(2)*G23','G33'],
            gram_block_sizes=[3]*(2*M),current_order='S0 nodes, then P1 nodes',ff_cap_cone_sizes=[3]*count,
            source_physical_rows=N,ordinary_scattering_rows_outside_current=N-2*M,
            scattering_policy='All scattering rows retained; current principal blocks imply S0/P1 disks',
            form_factor_rule='F=1+q(s)@ImF; q uses all sine-cardinal modes and q(0)=0; native ReF=1+K@ImF',
            spectrum_rule='Independent current spectra at M nodes; no off-node interpolation',
            source='v3 Eq.(2.33),(2.35)-(2.41),(3.65)-(3.75); K is Eq.(3.67)',uv=uv,
            scope='D1/D2 operators only; no joint feasibility/support claim',
            arithmetic='Arb K/k rounded to float64; amplitude rows unchanged'))

        if phase:

            def crossing(a,b):

    out=Path(args.output)

def resolution_difference(p,ref):
    key=(p['M'],p['L'])
    x=p['energy_gev']
    mask=(x>=ref['energy_gev'][0])&(x<=ref['energy_gev'][-1]);xx=x[mask];rphase=np.column_stack([np.interp(xx,ref['energy_gev'],ref['phase_degrees'][:,j]) for j in range(3)])
    diff=p['phase_degrees'][mask]-rphase;S=p['eta'][mask]*np.exp(2j*np.radians(p['phase_degrees'][mask]));Sr=ref['eta']*np.exp(2j*np.radians(ref['phase_degrees']))
    interp=np.column_stack([np.interp(xx,ref['energy_gev'],Sr[:,j].real)+1j*np.interp(xx,ref['energy_gev'],Sr[:,j].imag) for j in range(3)])
    return dict(M=key[0],L=key[1],common_native_grid=key[0]==50,energy_gev=xx,max_phase_difference_degrees=np.max(abs(diff),axis=0),max_mod_pi_difference_degrees=np.max(abs((diff+90)%180-90),axis=0),max_complex_S_difference=np.max(abs(S-interp),axis=0))
