def test_eq250_moments_from_independent_complex_ope_contour():
    from scipy.integrate import quad
    config = UVConfig(moment_source='eq250')
    pi, s0 = np.pi, config.s0;g = .023/.140**4
    j = -.0001/.140**4;moments = []
    for channel, n in [('S0', 0), ('S0', 1), ('P1', -1), ('P1', 0)]:
        def integrand(phi):
            s = s0*np.exp(1j*phi)
            logarithm = 1j*(phi-pi)  # log(-s/s0), with the physical positive-s cut.
            if channel == 'S0':
                polarization = 2*((.004+.0073)/(.140*2))**2/(2*pi)**4*(
                    -3/(8*pi*pi)*(1+13*.4/(3*pi))*s*logarithm-g/(8*s)-3*j/(2*s))
            else:
                polarization = .5/(2*pi)**4*(
                    -(1+.4/pi)*s*logarithm/(4*pi*pi)+g/(12*s)+j/s)
            return -np.exp(1j*(n+1)*phi)*polarization/s0
        real = quad(lambda phi: integrand(phi).real, 0, 2*pi, epsabs=1e-14)[0]
        imag = quad(lambda phi: integrand(phi).imag, 0, 2*pi, epsabs=1e-14)[0]
        moments.append(real+1j*imag)
    np.testing.assert_allclose(moments, fesr_targets(config), rtol=1e-12, atol=1e-22)

@pytest.mark.parametrize('chiral_norm,factor',[('separate-l2',1.),('combined-l2',3.)])
def test_joint_audit_restores_only_free_spectra_and_checks_the_complete_problem(chiral_norm,factor):
    """Synthetic M4 moments/caps from a known point; not a QCD feasible solution."""
    M=4;L=1;p=len(density_labels(M));n=3*M
    with ctx.workprec(256):s=np.repeat([float(x) for x in midpoint_grid(M)[0]],3)
    waves=np.tile([[0,0],[1,1],[2,0]],(M,1));kap=np.pi*np.sqrt(1-4/s)
    H=np.zeros((2*n+11,p));c=np.zeros(p);c[1+2*M:1+2*M+n]=1
    for j in range(M):
        H[3*j,0]=2.5;H[3*j+2,0]=1.;H[n+3*j,1+j]=1.5
        H[n+3*j,1+M+j]=1.;H[n+3*j+2,1+M+j]=1.
    for i in range(n):H[n+i,1+2*M+i]=1/kap[i]
    H[-2,1+2*M]=1.;current=current_operators(H,kap,s,waves,M);a=current['arrays']
    rho=3*a['k_squared'];rho[:,-1]=0;point=np.r_[c,np.zeros(2*M),rho.ravel()]
    a['moment_targets_raw']=current['matrices']['moment_linear']@point
    a['moment_errors_raw']=np.full(4,1e-8);a['ff_caps_squared']=4*a['k_squared'][:,-1]
    opts=dict(chiral_norm=chiral_norm,objective=None if factor==1 else factor*H[-2])
    complete,audit=joint_audit(H,kap,current,point,M,L,10.,.002,[1,0],**opts)
    assert len(complete)==p+4*M and np.array_equal(complete[:p+2*M],point[:p+2*M])
    assert audit['joint_primal_feasible'] and audit['lower']<=factor<=audit['upper']
    assert len(audit['restored_high_spectra'])==2 and len(audit['gram_principal_minor_enclosures'])==2*M
    final_rho=complete[p+2*M:].reshape(2,M);np.testing.assert_array_equal(final_rho[:,:-1],rho[:,:-1])
    np.testing.assert_allclose(final_rho[:,-1],2*a['k_squared'][:,-1],rtol=2e-10,atol=0)
    larger=complete.copy();larger[p+3*M-1]=1e6;larger[p+4*M-1]=2e6
    restored,_=joint_audit(H,kap,current,larger,M,L,10.,.002,[1,0],**opts);np.testing.assert_array_equal(restored[[p+3*M-1,p+4*M-1]],larger[[p+3*M-1,p+4*M-1]])
    bad=point.copy();bad[p+2*M]=-1
    _,failed=joint_audit(H,kap,current,bad,M,L,10.,.002,[1,0],**opts)
    assert failed['amplitude_primal_feasible'] and not failed['joint_primal_feasible'] and failed['lower'] is None
    rng=np.random.default_rng(19);G=rng.normal(size=(2,M,3,3))
    duals=dict(gram=G,gram_congruence=np.ones((2,M,3)),moment=rng.normal(size=4),ff=rng.normal(size=(2,1,2)))
    _,certificate=joint_audit(H,kap,current,point,M,L,10.,.002,[1,0],duals,**opts)
    assert certificate['primal_feasible'] and certificate['lower']<=factor<=certificate['upper']
    assert max(certificate['gram_dual_diagonal_shifts'])>0
    assert np.linalg.eigvalsh(np.asarray(certificate['repaired_normalized_gram_dual'])).min()>=-1e-12

@pytest.mark.parametrize('norm,width',[('separate-l2',4),('combined-l2',8)])
def test_chiral_group_barrier_curvature_matches_independent_log_ball(norm,width):
    z=np.array([.13,.17,.03,.02],np.longdouble);v=np.array([.3,-.2,.1,.4],np.longdouble)
    C=np.asarray(np.random.default_rng(51).normal(size=(8,4))*.2,np.longdouble)
    R=np.array([[.2,.1,0,0]]);I=np.array([[0,1.,0,0]]);t=np.ones(1)
    full=barrier_terms(z,R,I,t,C,.5,2.,2,chiral_norm=norm);base=barrier_terms(z,R,I,t,C,None,2.,2)
    F,b=full['features'],full['rhs'];G,d=base['features'],base['rhs'];gradient=-F.T@b+G.T@d;hessian=F.T@F-G.T@G
    value=lambda point:-np.log(np.longdouble('.25')-np.sum((C@point).reshape(-1,width)**2,axis=1)).sum()
    h=np.longdouble('1e-4');plus,minus=value(z+h*v),value(z-h*v)
    np.testing.assert_allclose(v@gradient,(plus-minus)/(2*h),rtol=1e-7,atol=1e-10)
    np.testing.assert_allclose(v@hessian@v,(plus-2*value(z)+minus)/(h*h),rtol=1e-6,atol=1e-9)
    assert len(full['chiral_slack'])==8//width

def test_resolution_projection_preserves_low_modes_and_actual_density_packing(target_M):
    """Nodal initialization only: no PV amplitude identity or feasibility transfer."""
    from smatrix_bootstrap.operators import project_coefficients
    def exact(M):
        phi=(np.arange(M)+.5)*np.pi/M;u,v=np.sin(phi),np.sin(2*phi)
        r1=np.outer(u,v);r2=np.outer(u,u)+(np.outer(u,v)+np.outer(v,u))/2
        return np.r_[.125,u,v,r1.ravel(),[r2[i,j]*(1 if i==j else 2) for i in range(M) for j in range(i,M)]]
    original=exact(4);projected=project_coefficients(original,target_M)
    assert np.shape(projected)==np.shape(exact(target_M))
    if target_M==4:assert np.array_equal(projected,original) and projected is not original
    else:np.testing.assert_allclose(projected,exact(target_M),rtol=3e-14,atol=3e-14)
    np.testing.assert_array_equal(original,exact(4))
def test_joint_figures_reject_identical_samples_from_different_UV_models(tmp_path,monkeypatch):
    from smatrix_bootstrap import io, model
    from types import SimpleNamespace
    from smatrix_bootstrap.analysis import gauge_regions, gauge_phases
    sig=dict(M=1,L=1,density_limit=1.,infinity='free',unitarity_scope='sampled',prescription='pv-midpoint',
        preparation=str(tmp_path/'H'),scattering_samples=3,chiral_norm='combined-l2')
    square=[[-1.,-1.],[1.,-1.],[1.,1.],[-1.,1.]];baseline=tmp_path/'IR.json'
    write_json(baseline,dict(model_signature=sig,regions=[dict(mode='chiral',epsilon=.002,points=[],
        geometry=dict(inner_vertices=square,outer_vertices=square,outer_closed=True))]))
    paths=[]
    for j,cutoff in enumerate(('hard-midpoint','clipped-phi')):
        path=tmp_path/f'UV{j}';path.mkdir();paths.append(path)
        report=dict(sig,mode='gauge',chiral_tolerance=.002,joint_feasible=True,all_original_constraints_checked=True,
            parameters=dict(preparation=sig['preparation'],current_preparation=str(tmp_path/'current'),direction=[1.,0.]),
            current_model=dict(uv=dict(config=dict(cutoff=cutoff))),solver=dict(all_amplitude_directions_retained=True),
            targets=[0.,0.],outer=dict(primal_feasible=True,joint_primal_feasible=True,lower=0.,upper=.1,enclosure='.1'))
        write_json(path/'report.json',report)
        for name in ('coefficients.json','current.json','joint.npz'):(path/name).touch()
    monkeypatch.setattr(io,'plot_gauge_region',lambda *a:None)
    result=gauge_regions(SimpleNamespace(output=tmp_path/'regions',region_summary=baseline,support_runs=paths))
    assert result['qualified_support_points']==1 and len(result['rejected'])==1
    assert result['rejected'][0]['reason']=='Joint input mismatch'
    assert result['model_signature']['uv_config']['cutoff']=='hard-midpoint'
    monkeypatch.setattr(model,'watson_diagnostics',lambda d:None)
    profiles=[]
    for j,role in enumerate(('tip','mid','ref')):
        path=tmp_path/role;path.mkdir();profiles.append(path)
        signature=dict(result['model_signature'],uv_config=dict(cutoff='clipped-phi' if j==1 else 'hard-midpoint'))
        write_json(path/'evaluation.json',dict(energies=[4.,5.,6.],waves=[[0,0],[2,0],[1,1]],
            f=np.zeros((3,3,2)),unitarity=dict(status='sampled_passed'),maximum_eta=1.,coefficients=str(path/'coefficients.json'),
            selection=dict(role=role,model_signature=signature,coefficients_mixed=False)))
    with pytest.raises(ValueError,match='same-model selections'):
        gauge_phases(SimpleNamespace(output=tmp_path/'phases',profile_runs=profiles))
def test_extra_analytic_rows_keep_native_current_identity_and_match_full_integral(tmp_path):

    from types import SimpleNamespace
    from smatrix_bootstrap.basis import prepare_cardinal,CardinalSourceRows
    from smatrix_bootstrap.sampling import prepared_sampling
    from smatrix_bootstrap.analytic import CardinalAmplitude
    from smatrix_bootstrap.run import support_data

    base=tmp_path/'base';extra=tmp_path/'extra';base.mkdir();extra.mkdir()
    a=SimpleNamespace(nodes=3,waves=1,bits=256,angular_order=24,subtracted=False,source_registry=None,preparation=None,
        output=base,density_limit=1500.,unitarity_energies=[],primary_waves=True,prescription='analytic-cardinal',command='prepare',infinity='free',unitarity_scope='sampled',additional_constraints=None)
    write_json(base/'report.json',prepare_cardinal(a,lambda **kw:None))
    a.preparation=base;a.output=extra;a.unitarity_energies=[9.5]
    write_json(extra/'report.json',prepare_cardinal(a,lambda **kw:None))
    with np.load(base/'amplitude.npz') as q:H0=q['rows'].copy()
    with np.load(extra/'amplitude.npz') as q:H=q['rows'].copy();E=q['radius_upper'].copy();ss,ww=prepared_sampling(q,3,1)

    a.preparation=extra;out=support_data(a,lambda **kw:None);np.testing.assert_array_equal(out[2],ss)
    current=current_operators(H,out[1],ss,ww,3);np.testing.assert_array_equal(current['arrays']['native_scattering_indices'],[[0,3,6],[1,4,7]])
    c=np.random.default_rng(37).normal(size=22);amp=CardinalAmplitude(c,3)
    with ctx.workprec(256):
        for j,(I,ell) in enumerate(ww[9:]):
            value=acb(sum((arb(float(v),float(e))*arb(float(x)) for v,e,x in zip(H[9+j],E[9+j],c)),arb(0)),sum((arb(float(v),float(e))*arb(float(x)) for v,e,x in zip(H[21+j],E[21+j],c)),arb(0)))
            assert (value-amp.partial_wave(arb('9.5'),None,int(I),int(ell))).contains(0)
    source=CardinalSourceRows(3,1,256);source.attach_preparation(extra)
    assert len(source.row(source.x[0],0,0,node=0))==22
    from smatrix_bootstrap.operators import PhaseOneProblem
    from smatrix_bootstrap.endpoints import configure_endpoint_coordinates
    v=np.sin((np.arange(3)+.5)*np.pi/3);c=np.r_[0.,v/8,v/8,np.outer(v,v).ravel(),[v[i]*v[j]/(2 if i==j else 1) for i in range(3) for j in range(i,3)]]*1e-4

    configure_endpoint_coordinates(P);z=P.initial;raw=P.raw(z)
    np.testing.assert_allclose(P.A@z[:22],H@raw[:22],rtol=1e-10,atol=1e-12)
    np.testing.assert_allclose(P.C@z[:22],H[24:32]@raw[:22],rtol=1e-10,atol=1e-12)
    D=P.congruence[P.keep];scale=np.column_stack((D[:,0]**2,D[:,0]*D[:,1],D[:,1]**2,D[:,0]*D[:,2],D[:,1]*D[:,2],D[:,2]**2));physical=(cur['arrays']['gram_constant']+cur['matrices']['gram_linear']@raw).reshape(6,6)[P.keep]

def test_center_acceptance_rejects_activity_changes_but_allows_free_spectral_lift(tmp_path):
    from smatrix_bootstrap.gauge import center_acceptance;from scipy.sparse import csr_matrix;from smatrix_bootstrap.ir import center_identity
    original=np.arange(9,dtype=float);current={'arrays':{'high_energy_indices':np.array([0])},'matrices':{'moment_linear':csr_matrix((4,9))}}
    lifted=original.copy();lifted[-2:]+=1;assert center_acceptance(original,lifted,1,current)['eligible']
    for slot in [0,4,5,6]:
        changed=lifted.copy();changed[slot]=np.nextafter(changed[slot],np.inf)
        assert not center_acceptance(original,changed,1,current)['eligible']
    np.savez(tmp_path/'center_converged.npz',has_newton_gradient=True,coefficients=original[:5],mu=1e-7);assert center_identity(tmp_path,original[:5],1,np.ones(2))['full_C_identical']
    with pytest.raises(ValueError,match='differs'):center_identity(tmp_path,original[:5]*(1-1e-8),1,np.ones(2))
def test_exact_form_factor_endpoint_and_dual_constant_identity():
    from smatrix_bootstrap.endpoints import complete_imf,endpoint_residual,asymptotic_audit,asymptotic_margins
    with ctx.workprec(256):
        b=[(arb(2*j+1)/8).tan_pi()/2 for j in range(2)]
        v=complete_imf([arb(2),arb(999)],True)
        expected=(1-(arb(2).sqrt()-1))/( (arb(2).sqrt()+1)/2 )
        assert (v[1]-expected).contains(0) and (1-sum(x*y for x,y in zip(b,v))).contains(0)
        r=[arb('0.3'),arb('-0.7')];q,constant=endpoint_residual(r,b)
        assert q[-1].is_zero() and (sum(x*y for x,y in zip(r,v))-constant-sum(x*y for x,y in zip(q,v))).contains(0)
        assert complete_imf([arb(2),arb(999)],False)[1]==999
        c=np.arange(12)/10;_,support,m=asymptotic_audit(c,2,{'asymptotic_kR':[0,0,1,0,0],'asymptotic_kI':[0,0,-2,0,0]});assert support==arb(1)/4 and all((x*a-y).contains(0) for x,a,y in zip(m,[1,1,5,3,5],asymptotic_margins(c,2)))
